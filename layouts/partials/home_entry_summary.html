` $prose -}} {{- if . | plainify | strings.TrimSpace -}} {{- $preview = . -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- $text := replaceRE `\s+` " " ($preview | plainify | htmlUnescape) | strings.TrimSpace -}} {{- return (strings.Truncate 120 $text) -}}
